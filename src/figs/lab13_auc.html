from scipy.integrate import trapezoid

rows = []
for sid, g in theo.groupby("Subject"):
    g = g.sort_values("Time")
    rows.append({"Subject": sid,
                 "AUC_0_t": trapezoid(g["conc"], g["Time"]),
                 "Cmax": g["conc"].max(),
                 "Tmax": g.loc[g["conc"].idxmax(), "Time"]})
pk = pd.DataFrame(rows)
pk.round(2)

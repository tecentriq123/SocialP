def beta_ab(m, se):
    """평균, 표준오차 -> 베타분포의 alpha, beta"""
    n = m * (1 - m) / se ** 2 - 1
    return m * n, (1 - m) * n

def gamma_ks(m, se):
    """평균, 표준오차 -> 감마분포의 모양, 척도"""
    shape = (m / se) ** 2
    return shape, m / shape

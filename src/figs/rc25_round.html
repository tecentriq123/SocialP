round(c(2.5, 3.5, 0.5))              # 정확히 반인 값은 짝수 쪽으로
round(56.25, 1); round(0.125, 2)
sprintf("%.1f", 56.25)               # 글자로 바꿀 때도 같음
half_up <- function(x, digits = 0) floor(x * 10^digits + 0.5) / 10^digits   # 엑셀 ROUND처럼 올림
half_up(c(2.5, 56.25, 0.125), c(0, 1, 2))

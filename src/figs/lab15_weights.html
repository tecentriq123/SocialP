pA = df["drugA"].mean()                   # A를 받은 비율 0.4053
ps = df["ps"]
df["w"] = np.where(a, 1 / ps, 1 / (1 - ps))            # IPTW
df["sw"] = np.where(a, pA / ps, (1 - pA) / (1 - ps))   # 안정화

print(df[["w", "sw"]].agg(["mean", "min", "max"]).round(3))
print(df.groupby("drugA")[["w", "sw"]].sum().round(0))

print("평균 x 인원   :", round(b.mean() * len(b)))
print("중앙값 x 인원 :", round(np.median(b) * len(b)))
print("평균보다 적게 쓴 환자(%):",
      round((b < b.mean()).mean() * 100, 1))
k = round(len(b) * 0.05)                          # 상위 5%는 90명
top = np.sort(b)[::-1][:k]
print("상위 5%가 쓴 몫(%)      :",
      round(top.sum() / b.sum() * 100, 1))

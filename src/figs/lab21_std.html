      round(pB.mean(), 1), round(pA.mean() - pB.mean(), 1))
print("비:", round(pA.mean() / pB.mean(), 3))

d_i = pA - pB                        # 환자 한 명 한 명의 예측 차이
print("환자별 차이의 범위:", round(d_i.min(), 1), "~",

long["py"] = (long["stop"] - long["start"]) / 365
t3 = long.groupby("drugx").agg(deaths=("died", "sum"),
                               py=("py", "sum"))
t3["rate"] = t3["deaths"] / t3["py"] * 100      # 100인년당 사망률
print("발생률비:", round(t3.loc[1, "rate"] / t3.loc[0, "rate"], 2))
t3.round({"py": 0, "rate": 1})

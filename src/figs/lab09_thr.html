def classify(t):
    pred = (p_test >= t).astype(int)         # 임계값 t 이상 = 악성
    tn, fp, fn, tp = confusion_matrix(y_test, pred).ravel()
    return {"threshold": t, "TP": tp, "FN": fn, "FP": fp, "TN": tn,
            "sensitivity": tp / (tp + fn),
            "specificity": tn / (tn + fp),
            "accuracy": (tp + tn) / len(pred)}

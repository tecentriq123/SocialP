def path_locf(month, u, t_death, end=24):
    """마지막 측정값이 사망 직전까지 유지된다고 보는 경로"""
    t, u = list(month), list(u)
    if t_death <= end:                    # 사망 직전까지 마지막 값
        t, u = t + [t_death, t_death], u + [u[-1], 0.0]
        if t_death < end:
            t, u = t + [end], u + [0.0]
    return np.array(t, dtype=float), np.array(u, dtype=float)

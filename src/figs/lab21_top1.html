k1 = round(len(b) * 0.01)                 # 상위 1%는 18명
srt = np.sort(b)[::-1]                    # B군 비용을 큰 값부터
print("상위 1% 인원      :", k1, "명")
print("뺀 환자의 최소 비용:", srt[k1 - 1])
print("B군 평균          :", round(b.mean(), 1))
print("상위 1%를 뺀 평균 :", round(srt[k1:].mean(), 1))
print("상위 1%가 쓴 비율(%):", round(srt[:k1].sum() / b.sum() * 100, 1))

def or_per(m, var, c):                  # c 단위당 오즈비와 95% CI
    est = m.params[var]                 # 1단위당 로그 오즈비
    lo, hi = m.conf_int().loc[var]      # 그 95% 신뢰구간
    return np.round(np.exp([c * est, c * lo, c * hi]), 3)

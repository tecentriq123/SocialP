ad <- its[!(its$time %in% 37:39), ]                         # 시행 직후 3개월을 뺌
fa <- lm(rate ~ time + post + time_after, data = ad)
round(cbind(coef = coef(fa), coefci(fa, vcov. = NeweyWest(fa, lag = 3, prewhite = FALSE), df = Inf))[c("post", "time_after"), ], 3)
nw(fa)                                                      # 12개월째

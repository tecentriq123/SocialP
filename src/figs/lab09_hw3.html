keep = []
for t in terms:                                 # 셀 8의 일곱 변수
    m1 = smf.logit(f"low ~ {t}", data=bw).fit(disp=0)
    p1 = m1.pvalues.drop("Intercept").min()     # 가변수가 둘이면 작은 쪽
    print(f"{t:8} univariable P = {p1:.3f}")
    if p1 < 0.05:
        keep.append(t)

f_sel = "low ~ " + " + ".join(keep)            # 동료가 고른 모형

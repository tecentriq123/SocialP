def summary(c):
    t = rate_table(c)
    out = {}
    for arm in ["SGLT2i", "DPP4i"]:
        r = t.loc[arm]
        out[arm + " events / PY"] = "%d / %.0f" % (r["events"],
                                                   r["py"])
        out[arm + " rate (95% CI)"] = "%.2f (%.2f-%.2f)" % (
            r["rate"], r["lo"], r["hi"])
    out["IRR (95% CI)"] = "%.2f (%.2f-%.2f)" % irr(t)
    out["crude HR (95% CI)"] = "%.2f (%.2f-%.2f)" % hr(c, ["sglt2"])
    out["adjusted HR (95% CI)"] = "%.2f (%.2f-%.2f)" % hr(
        c, ["sglt2"] + covs)
    return out

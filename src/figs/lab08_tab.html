셀 4흡연과 저체중아의 2×2 표
tab = pd.crosstab(df["smoke"], df["low"]).loc[[1, 0], [1, 0]]
print(tab)
(a, b), (c, d) = tab.to_numpy()
p1, p0 = a / (a + b), c / (c + d)
print(f"risk: smoker = {p1:.3f}, nonsmoker = {p0:.3f}")

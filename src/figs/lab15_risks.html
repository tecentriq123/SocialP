def risks(d, w=None):
    """A군 위험, B군 위험, 상대위험도, 위험차 (w는 가중치 열 이름)."""
    t = d["drugA"] == 1
    w = np.ones(len(d)) if w is None else d[w].to_numpy()
    rA = np.average(d.loc[t, "hosp_1y"], weights=w[t])
    rB = np.average(d.loc[~t, "hosp_1y"], weights=w[~t])
    return [rA, rB, rA / rB, rA - rB]

셀 11시점별 생존율과 95% 신뢰구간
ci = kmf.confidence_interval_
for t in [6, 12, 24]:
    s = kmf.survival_function_at_times(t).iloc[0]
    lo, hi = ci.loc[:t].iloc[-1]          # t 이전 마지막 계단
    print(f"{t:>2}개월 생존율 {s:.3f} (95% CI {lo:.3f}-{hi:.3f})")

    pd.DataFrame({"t": u["fu_days"] - u["rx_day"],   # 첫 처방일부터
                  "died": u["died"], "ever": 1}),
    pd.DataFrame({"t": nu["fu_days"],                # 퇴원일부터
                  "died": nu["died"], "ever": 0})])

      " 발생률비:", round(rate2[1] / rate2[0], 2))

rng = np.random.default_rng(21)
idx = rng.integers(0, len(a), len(a))    # 0~1202 가운데 1,203개
print(idx[:8])
print("한 번 이상 뽑힌 환자:", len(np.unique(idx)), "명")
print("원래 A군의 평균     :", round(a.mean(), 1))
print("다시 뽑은 A군의 평균:", round(a[idx].mean(), 1))

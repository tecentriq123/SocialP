def hf_share(d, w=None):
    """군별 (가중) 심부전 비율 [A군, B군]. w는 가중치 열 이름."""
    t = d["drugA"] == 1
    w = np.ones(len(d)) if w is None else d[w].to_numpy()
    return [np.average(d.loc[t, "hf"], weights=w[t]),
            np.average(d.loc[~t, "hf"], weights=w[~t])]

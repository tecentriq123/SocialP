occupancy <- function(pfs, os) {
  pf <- pmin(pfs, os)                    # PFS가 OS를 넘지 않게
  data.frame(pf = pf, pd = os - pf, dead = 1 - os)
}
occ_B <- occupancy(pfs_B, os_B)
occ_A <- occupancy(pfs_A, os_A)
dim(occ_B)
round(unlist(occ_B[24 + 1, ]), 4)        # 24개월(25번째 행)

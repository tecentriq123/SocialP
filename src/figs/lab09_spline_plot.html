import matplotlib.pyplot as plt

grid = pd.DataFrame({"lwt": np.arange(85, 251, 5)}).assign(
    age=23, race="white", smoke=0, ptd=0, ht=0, ui=0)
def log_odds(m):
    p = m.predict(grid)                 # 예측 확률
    return np.log(p / (1 - p))          # 확률 -> 로그 오즈

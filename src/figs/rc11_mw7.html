A <- c(1, 2, 3, 3, 4, 5, 47); B <- 6:12       # 1장 g 절의 재원일수 7명씩
wilcox.test(A, B, exact = FALSE)$p.value                    # 정규근사(동점 보정, 연속성 보정)
wilcox.test(A, B, exact = FALSE, correct = FALSE)$p.value   # 연속성 보정 없이

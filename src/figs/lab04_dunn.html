mean_rank = df.groupby("race_f", observed=True)["rank"].mean()

p_dunn = []
for a, b in pairs:
    se = np.sqrt((N * (N + 1) / 12 - tie) * (1 / n[a] + 1 / n[b]))
    z = (mean_rank[a] - mean_rank[b]) / se
    p_dunn.append(2 * stats.norm.sf(abs(z)))
p_dunn_holm = multipletests(p_dunn, method="holm")[1]

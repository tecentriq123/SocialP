def gap(r, arm, target):
    """진행 -> 사망 율이 r일 때 모형의 중앙 OS - 목표 중앙값"""
    h = hr if arm == "A" else 1.0
    return median_os(matrix(0.9 * r_exit * h, 0.1 * r_exit * h, r)) - target

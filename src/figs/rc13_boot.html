suppressPackageStartupMessages(library(boot))
set.seed(2026)
d8 <- data.frame(age, sbp)
b <- boot(d8, function(d, i) cor(d$age[i], d$sbp[i]), R = 4000)   # 환자를 짝지은 채 다시 뽑기
boot.ci(b, type = c("perc", "bca"))                 # 백분위수 구간과 BCa 구간

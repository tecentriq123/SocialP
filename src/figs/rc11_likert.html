control <- rep(1:5, times = c(2, 5, 7, 4, 2))   # 대조군 20명
wilcox.test(counsel, control, exact = FALSE)$p.value     # 동점 보정 + 연속성 보정
wilcox.test(counsel, control, exact = FALSE, correct = FALSE)$p.value

s1, s2 = wide["drug1"].std(), wide["drug2"].std()
sd_paired = np.sqrt(s1**2 + s2**2 - 2 * r * s1 * s2)
sd_indep = np.sqrt(s1**2 + s2**2)            # r = 0으로 본 경우
print(f"SD1 = {s1:.3f}, SD2 = {s2:.3f}, r = {r:.3f}")
print(f"SD of d: paired {sd_paired:.3f}, "
      f"if r = 0 {sd_indep:.3f}")

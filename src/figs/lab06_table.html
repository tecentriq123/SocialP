def tidy(m):
    ci = m.conf_int()
    f2 = "{:.2f}".format
    est = (m.params.map(f2) + " (" + ci[0].map(f2) + " to "
           + ci[1].map(f2) + ")")
    p = m.pvalues.map(lambda x: "<0.001" if x < 0.001 else f"{x:.3f}")
    return pd.DataFrame({"beta (95% CI)": est, "P": p})

terms = ["bmi", "age", "C(sex)", "bp", "s5"]

for name, k in km_sex.items():

    rows.append({"group": name,
                 "n": k.event_table["at_risk"].iloc[0],
                 "deaths": k.event_table["observed"].sum(),
                 "median (95% CI)": f"{med:.1f} ({lo:.1f}-{hi:.1f})",
                 "1-yr % (95% CI)": f"{s12:.1f} ({l12:.1f}-{h12:.1f})"})
pd.DataFrame(rows)

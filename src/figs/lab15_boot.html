    p = smf.logit(f_ps, data=d).fit(disp=0).predict(d)  # 성향점수부터 다시
    d = d.assign(wb=np.where(d["drugA"] == 1, 1 / p, 1 / (1 - p)))
    boot.append(risks(d, "wb")[2])                    # 가중 RR
print("bootstrap 95% CI:", np.quantile(boot, [0.025, 0.975]).round(3))

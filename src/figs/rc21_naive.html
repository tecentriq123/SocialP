welch <- function(x, post) {                     # 시행 전후 평균의 Welch t 검정
  t <- t.test(x[post == 1], x[post == 0])
  round(c(pre = mean(x[post == 0]), post = mean(x[post == 1]),
          diff = unname(t$estimate[1] - t$estimate[2]), lower = t$conf.int[1], upper = t$conf.int[2]), 2)
}

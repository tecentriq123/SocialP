nw <- function(f, L = c(0, 0, 1, 12)) {                     # 계수 조합과 Newey–West 95% CI
  V <- NeweyWest(f, lag = 3, prewhite = FALSE)
  e <- sum(L * coef(f)); se <- sqrt(drop(t(L) %*% V %*% L))
  round(c(est = e, lower = e - 1.96 * se, upper = e + 1.96 * se), 3)
}

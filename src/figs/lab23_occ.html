def occupancy(p, arm, t):
    """시점별 상태 비율: 무진행(pf), 진행(pd), 사망(dead)"""
    s_pfs, s_os = curves(p, arm, t)
    s_pfs = np.minimum(s_pfs, s_os)    # PFS가 OS를 넘지 않게
    return pd.DataFrame({"pf": s_pfs, "pd": s_os - s_pfs,
                         "dead": 1 - s_os}, index=t)

occ_A = occupancy(p, "A", t)

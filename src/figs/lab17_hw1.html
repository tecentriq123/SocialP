for g in ["A", "B"]:
    d = ckd[ckd["drug"] == g]
    est, lo, hi = cif(d, 2, 5)             # 관심 사건 = 사망(2)
    km = KaplanMeierFitter().fit(d["years"], d["status"] == 2)
    print(f"{g}: CIF {est*100:.1f} ({lo*100:.1f}-{hi*100:.1f}),"
          f" 1 - KM {(1 - km.predict(5))*100:.1f}")

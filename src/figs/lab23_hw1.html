for years in [3, 5, 10, 20]:
    r = run_model(p, horizon=years * 12)

                   * (years * 12) ** p["os_gam"])
    rows.append([years, r["d_cost"], r["d_qaly"], r["icer"], alive])
hw1 = pd.DataFrame(rows, columns=["years", "d_cost", "d_qaly",
                                  "icer", "alive_A"])

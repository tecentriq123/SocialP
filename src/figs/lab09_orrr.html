tab = pd.crosstab(bw["smoke"], bw["low"])   # 행 smoke, 열 low
risk = tab[1] / tab.sum(axis=1)             # 저체중아 비율(위험)
odds = tab[1] / tab[0]                      # 오즈
print(tab)
print("risk:", risk.round(3).tolist())
print("RR =", round(risk[1] / risk[0], 2),
      "  OR =", round(odds[1] / odds[0], 2))

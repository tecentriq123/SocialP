print(f"I2 = {m['I2']:.0%}")

print(f"exp(mean +/- 1.96 tau): {np.exp(m['re'] - Z*tau):.2f}"
      f"-{np.exp(m['re'] + Z*tau):.2f}")
print(f"t({m['k'] - 2}) = {tcrit:.3f}, half-width = {half:.3f}")

      f"-{np.exp(m['re'] + half):.2f}")

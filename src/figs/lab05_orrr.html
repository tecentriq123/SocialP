셀 11오즈비와 상대위험도의 관계
rr, or_ = t22.riskratio, t22.oddsratio
print(f"RR = {rr:.3f}, OR = {or_:.3f}")
print("RR x (1-p0)/(1-p1) =", round(rr * (1 - p0) / (1 - p1), 3))
print("OR/(1-p0+p0*OR) =", round(or_ / (1 - p0 + p0 * or_), 3))

                   "exp(coef) upper 95%", "p"]].copy()

tab["HR (95% CI)"] = [f"{h:.2f} ({l:.2f}-{u:.2f})"
                      for h, l, u in zip(tab.HR, tab.lower, tab.upper)]
tab["P"] = [f"{p:.3f}" if p >= 0.001 else "<0.001" for p in tab.p]
tab[["HR (95% CI)", "P"]]

def three_tests(x):                    # 차이 x에 세 검정을 한 번에
    t = stats.ttest_1samp(x, 0)
    return {"mean": x.mean(), "SD": x.std(),
            "paired t p": t.pvalue,
            "signed-rank p": stats.wilcoxon(x).pvalue,
            "sign test p": stats.binomtest(int((x > 0).sum()),
                                           int((x != 0).sum())).pvalue}

d2 = d.copy()

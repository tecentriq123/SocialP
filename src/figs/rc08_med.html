median_os <- function(P, n = 1200) {            # 생존 비율이 0.5가 되는 시점(개월)
  alive <- rowSums(trace(P, n)[, c("pf", "pd")])
  k <- which(alive <= 0.5)[1]                     # 처음으로 0.5 이하가 된 행 (k행 = k - 1개월)
  (k - 2) + (alive[k - 1] - 0.5) / (alive[k - 1] - alive[k])
}

nz = d[d != 0]                   # 차이가 0인 쌍은 제외
rk = stats.rankdata(nz.abs())    # |d|에 순위 (동점은 평균순위)
print(pd.DataFrame({"d": nz, "rank": rk}).T)
w_plus = rk[nz > 0].sum()        # 늘어난 쪽의 순위합
w_minus = rk[nz < 0].sum()       # 줄어든 쪽의 순위합
print("n =", len(nz), " W+ =", w_plus, " W- =", w_minus)

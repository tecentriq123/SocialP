rm28 <- function(time, event, arm) {
  t <- summary(survfit(Surv(time, event) ~ arm), rmean = 28)$table[, "rmean"]
  round(c(B = unname(t[1]), A = unname(t[2]), diff = unname(t[2] - t[1])), 2)
}

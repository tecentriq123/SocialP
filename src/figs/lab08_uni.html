def or_ci(model):
    t = np.exp(model.conf_int())
    t.columns = ["lo", "hi"]
    t.insert(0, "OR", np.exp(model.params))
    return t.drop(index="Intercept")

terms = ["smoke", "C(race)", "age", "lwt_kg", "ptl_any", "ht"]
uni = pd.concat([or_ci(smf.logit(f"low ~ {t}", df).fit(disp=0))
                 for t in terms])
uni.round(2)

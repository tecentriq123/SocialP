def stop_date(own, grace):
    over = own["gap"].isna() | (own["gap"] > grace)
    runout = own[over].groupby("pid")["runout"].first()
    return runout + pd.Timedelta(days=grace)

cohort["stop_date"] = cohort["pid"].map(stop_date(own, 60))

library(broom)
hr <- tidy(fit, exponentiate = TRUE, conf.int = TRUE)
data.frame(term = hr$term,
           `HR (95% CI)` = sprintf("%.2f (%.2f-%.2f)", hr$estimate,
                                   hr$conf.low, hr$conf.high),
           P = format.pval(hr$p.value, digits = 2, eps = 0.001),
           check.names = FALSE)

셀 5군별 발생률과 정확 신뢰구간
from scipy import stats

rt = pt.groupby("treat", observed=True).agg(
    patients=("id", "size"),
    events=("n_inf", "sum"),
    py=("py", "sum"))
d, T = rt["events"], rt["py"]
rt["rate"] = d / T * 100                          # 100인년당
rt["lower"] = stats.chi2.ppf(0.025, 2 * d) / 2 / T * 100
rt["upper"] = stats.chi2.ppf(0.975, 2 * d + 2) / 2 / T * 100
rt.round(2)

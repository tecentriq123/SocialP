for g in ["A", "B"]:
    d = ckd[ckd["drug"] == g]
    km = KaplanMeierFitter().fit(d["years"], d["any"])
    both = cif(d, 1, 5)[0] + cif(d, 2, 5)[0]
    print(f"{g}: 1 - KM {(1 - km.predict(5))*100:.1f},"
          f" CIF dialysis + CIF death {both*100:.1f}")

def or_ci(m):
    ci = np.exp(m.conf_int())
    f2 = "{:.2f}".format
    return (np.exp(m.params).map(f2) + " (" + ci[0].map(f2)
            + " to " + ci[1].map(f2) + ")")

terms = ["age", "lwt", "C(race)", "smoke", "ptd", "ht", "ui"]

                   .drop("Intercept") for t in terms])

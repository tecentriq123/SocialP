def log_or(e1, n1, e0, n0):
    """로그 오즈비와 분산 (0인 칸이 있는 시험만 0.5씩 더함)"""
    a, c = e1.astype(float), e0.astype(float)
    b, d = n1 - a, n0 - c
    zero = (a == 0) | (b == 0) | (c == 0) | (d == 0)
    a, b, c, d = [x + 0.5 * zero for x in (a, b, c, d)]
    return np.log(a * d / (b * c)), 1/a + 1/b + 1/c + 1/d

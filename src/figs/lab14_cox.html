from lifelines import CoxPHFitter

def hr(c, cols):
    m = CoxPHFitter().fit(c[["time", "event"] + cols],
                          "time", "event")
    s = m.summary.loc["sglt2"]
    return (s["exp(coef)"], s["exp(coef) lower 95%"],
            s["exp(coef) upper 95%"])

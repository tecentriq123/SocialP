셀 11역 Kaplan-Meier로 중앙 추적기간 구하기
km <- survfit(Surv(time / 365.25, 1 - event) ~ 1, data = full)   # 사건과 중도절단을 뒤집은 곡선
quantile(km, 0.5)$quantile                                        # 역 Kaplan-Meier 중앙 추적기간(년)
median(full$time) / 365.25                                        # time의 단순 중앙값

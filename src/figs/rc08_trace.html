trace <- function(P, n = 240) {                  # 모두 무진행에서 출발해 n주기 추적
  x <- matrix(0, n + 1, 3, dimnames = list(NULL, colnames(P)))
  x[1, ] <- c(1, 0, 0)
  for (k in 1:n) x[k + 1, ] <- x[k, ] %*% P      # 행렬 곱 한 번 = 한 달
  x
}

L = 365 * 3 / 12
lm_w = d[d["fu_days"] > L].copy()          # 3개월 생존자
lm_w["t_lm"] = lm_w["fu_days"] - L
print(lm_w.groupby("ever")["died"].agg(["count", "sum"]))
m1 = CoxPHFitter().fit(lm_w, "t_lm", "died", formula="ever")
print("HR (95% CI):", hr_ci(m1, "ever"))

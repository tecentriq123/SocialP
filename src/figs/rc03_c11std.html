std_surv <- function(g, times) {          # 모두 약물 g를 받았다고 놓은 곡선의 평균
  nd <- transform(co, drug = factor(g, levels = c("B", "A")))
  rowMeans(summary(survfit(fm, newdata = nd), times = times)$surv)
}
at <- c(12, 24, 48)
round(rbind(A = std_surv("A", at), B = std_surv("B", at)), 3)

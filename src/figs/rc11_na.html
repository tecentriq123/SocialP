x_na <- c(8, 12, NA, 11, 10)              # 셋째 환자의 값이 빔
mean(x_na)                                # NA가 하나라도 있으면 NA
c(mean = mean(x_na, na.rm = TRUE), sd = sd(x_na, na.rm = TRUE),
  n = sum(!is.na(x_na)))
cost <- c(0, 0, 12, 18, 20)               # 1인당 비용(만 원)
c(all = mean(cost), drop_zero = mean(cost[cost > 0]))

pct_ci = (np.exp(model_log.conf_int()) - 1) * 100   # CI 양 끝 변환
out = pd.concat([model_log.params, pct, pct_ci], axis=1)
out.columns = ["beta (log)", "% diff", "lower %", "upper %"]
out.drop("Intercept").round({"beta (log)": 4, "% diff": 1,
                             "lower %": 1, "upper %": 1})

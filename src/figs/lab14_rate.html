from scipy import stats

def rate_table(c):
    t = c.groupby("arm").agg(n=("event", "size"),
                             events=("event", "sum"),
                             py=("time", "sum"))
    t["py"] = t["py"] / 365.25
    d = t["events"]
    t["rate"] = 1000 * d / t["py"]
    t["lo"] = 1000 * stats.chi2.ppf(0.025, 2 * d) / 2 / t["py"]
    t["hi"] = 1000 * stats.chi2.ppf(0.975, 2 * d + 2) / 2 / t["py"]
    return t

d <- 3; t <- 1500                                   # 혈관부종 3건 / 1,500인년
round(1000 * rbind(Wald  = d / t + c(-1, 1) * 1.96 * sqrt(d) / t,
                   log   = exp(log(d / t) + c(-1, 1) * 1.96 / sqrt(d)),
                   exact = poisson.test(d, t)$conf.int), 2)

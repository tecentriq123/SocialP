셀 20과제 1 정답. 반복 횟수와 몬테카를로 오차
ok = (5000 * dq - dc) > 0              # 벌마다 비용효과적인가
for n in [500, 1000, 5000]:
    pr = ok[:n].mean()                 # 처음 n벌만 쓴 확률
    se = np.sqrt(pr * (1 - pr) / n)    # 몬테카를로 표준오차
    print(n, " 확률", round(pr, 4), " 표준오차", round(se, 4),
          " 범위", round(pr - 1.96 * se, 3), "-",
          round(pr + 1.96 * se, 3))
